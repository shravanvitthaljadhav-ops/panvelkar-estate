function renderUnits(){ const q=(document.getElementById('unitSearch').value||'').trim().toLowerCase(); let units=state.units.filter(u=>u.unit_type==='flat'); if(q){ units=units.filter(u=>{ const wing=state.wings.find(w=>w.id===u.wing_id); const wingCode=wing?.code||''; return String(u.unit_number||'').toLowerCase().includes(q) || wingCode.toLowerCase().includes(q) || `${wingCode}-${u.unit_number}`.toLowerCase().includes(q); }); } const rows=units.map(u=>{ const wing=state.wings.find(w=>w.id===u.wing_id); const profile=state.profiles.find(p=>p.unit_id===u.id && p.active!==false); const tenant=state.tenants.find(t=>t.unit_id===u.id && t.active!==false); const vehicle=state.vehicles.find(v=>v.unit_id===u.id && v.active!==false); let resident='—'; if(u.occupied_status==='owner_occupied'){ resident=profile?.full_name||'Owner record not published'; }else if(u.occupied_status==='rented'){ resident=tenant?.full_name||'Tenant record not published'; } return ` ${esc(wing?.code||'—')} ${esc(u.unit_number)} ${esc(u.floor_number??'—')} ${esc(statusLabel(u.occupied_status))} ${esc(resident)} ${esc(vehicle?.vehicle_number||'—')} `; }).join(''); document.getElementById('unitRows').innerHTML= rows || 'No matching flats found.';
